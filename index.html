import React, { useEffect, useState } from "react";

/**
 * ARWaveApp.jsx
 * Single-file React component that embeds an A-Frame + AR.js scene (marker-based)
 * and provides a simple Tailwind-styled UI for controlling a procedural 3D wave.
 *
 * How this works (summary):
 * - A-Frame + AR.js are loaded dynamically via script tags.
 * - A custom A-Frame component `sine-wave` creates a THREE.BufferGeometry line/mesh
 *   representing a 1D wave and updates vertex positions on each tick based on
 *   amplitude & frequency props.
 * - The React UI updates amplitude/frequency/waveType state which in turn sets
 *   attributes on the A-Frame entities inside the AR markers.
 * - Two markers are provided (hiro preset and kanji preset) to represent
 *   transversal and longitudinal variants (multi-marker support).
 *
 * Notes:
 * - This implementation uses marker-based WebAR (AR.js) which runs fully in the
 *   mobile browser (no native app required). For best results use Chrome on Android
 *   or Safari on iOS (iOS 11+ with supported browsers).
 * - You will need to host/serve the pattern markers (or use the built-in presets 'hiro'/'kanji').
 * - This is a prototype scaffold to be adapted and hardened for production.
 */

export default function ARWaveApp() {
  const [amplitude, setAmplitude] = useState(0.5);
  const [frequency, setFrequency] = useState(1.0);
  const [waveType, setWaveType] = useState("transversal"); // 'transversal' | 'longitudinal'

  useEffect(() => {
    // Dynamically inject A-Frame and AR.js (aframe-ar) scripts if not already present
    const loadScript = (src) =>
      new Promise((resolve, reject) => {
        if (document.querySelector(`script[src='${src}']`)) return resolve();
        const s = document.createElement("script");
        s.src = src;
        s.async = true;
        s.onload = resolve;
        s.onerror = reject;
        document.head.appendChild(s);
      });

    (async () => {
      try {
        await loadScript("https://aframe.io/releases/1.4.0/aframe.min.js");
        await loadScript("https://raw.githack.com/AR-js-org/AR.js/master/aframe/build/aframe-ar.js");

        // register a custom component to draw/update a sine wave mesh
        if (window.AFRAME && !window.AFRAME.components["sine-wave"]) {
          window.AFRAME.registerComponent("sine-wave", {
            schema: {
              amplitude: { type: "number", default: 0.5 },
              frequency: { type: "number", default: 1.0 },
              length: { type: "number", default: 2.0 },
              points: { type: "int", default: 64 },
              type: { type: "string", default: "transversal" } // transversal or longitudinal
            },
            init: function () {
              const points = this.data.points;
              const geometry = new THREE.BufferGeometry();
              const positions = new Float32Array(points * 3);
              const indices = new Uint16Array((points - 1) * 2);

              for (let i = 0; i < points; i++) {
                const idx = i * 3;
                positions[idx] = (i / (points - 1) - 0.5) * this.data.length; // x
                positions[idx + 1] = 0; // y
                positions[idx + 2] = 0; // z
              }
              geometry.setAttribute("position", new THREE.BufferAttribute(positions, 3));

              // simple line material
              const material = new THREE.MeshStandardMaterial({ color: 0x1e90ff, side: THREE.DoubleSide });

              // create tube geometry from the path (for a ribbon-like wave)
              // We will update positions each frame
              this.geometry = geometry;
              this.positions = positions;

              // Create a simple mesh via THREE.Line
              const lineMaterial = new THREE.LineBasicMaterial({ color: 0x1e90ff, linewidth: 2 });
              const line = new THREE.Line(geometry, lineMaterial);

              // Store and add to entity
              this.el.object3D.add(line);
              this.line = line;
            },
            update: function () {
              // apply immediate update if attributes changed
              this.tick(0);
            },
            tick: function (time, timeDelta) {
              const amp = this.data.amplitude;
              const freq = this.data.frequency;
              const pts = this.data.points;
              const len = this.data.length;
              const type = this.data.type;

              const positions = this.geometry.attributes.position.array;

              for (let i = 0; i < pts; i++) {
                const idx = i * 3;
                const x = (i / (pts - 1) - 0.5) * len;
                // Sine argument includes time so it animates
                const phase = time * 0.001 * freq * 2.0 * Math.PI;
                const y = amp * Math.sin((i / (pts - 1)) * freq * 2.0 * Math.PI + phase);

                if (type === "transversal") {
                  positions[idx] = x; // x
                  positions[idx + 1] = y; // y
                  positions[idx + 2] = 0; // z
                } else {
                  // longitudinal: displacement along z axis, small y for visibility
                  positions[idx] = x; // x
                  positions[idx + 1] = 0.02 * Math.sin((i / (pts - 1)) * freq * 2.0 * Math.PI + phase); // slight y
                  positions[idx + 2] = amp * Math.sin((i / (pts - 1)) * freq * 2.0 * Math.PI + phase); // z
                }
              }

              this.geometry.attributes.position.needsUpdate = true;
            },
            remove: function () {
              // cleanup if needed
            }
          });
        }
      } catch (err) {
        console.error("Error loading AR dependencies:", err);
      }
    })();
  }, []);

  // Update A-Frame entities when React state changes
  useEffect(() => {
    const selector = waveType === "transversal" ? "#wave-transversal" : "#wave-longitudinal";
    const el = document.querySelector(selector);
    if (el) {
      el.setAttribute("sine-wave", `amplitude: ${amplitude}; frequency: ${frequency}; type: ${waveType}`);
    }
  }, [amplitude, frequency, waveType]);

  return (
    <div className="min-h-screen bg-gray-50 p-4">
      <div className="max-w-4xl mx-auto">
        <h1 className="text-2xl font-semibold mb-3">AR Gelombang Mekanik â Prototype WebAR</h1>

        <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
          {/* Left: AR scene */}
          <div className="bg-white shadow rounded p-2">
            <div style={{ width: "100%", height: "480px" }}>
              {/* A-Frame scene (marker-based WebAR) */}
              <div id="aframe-root">
                <a-scene
                  vr-mode-ui="enabled: false"
                  embedded
                  arjs="sourceType: webcam; debugUIEnabled: false; trackingMethod: best;"
                  style={{ width: "100%", height: "100%" }}
                >
                  <a-assets></a-assets>

                  {/* Lighting */}
                  <a-entity light="type: ambient; intensity: 0.8"></a-entity>
                  <a-entity light="type: directional; intensity: 0.6" position="0 1 0"></a-entity>

                  {/* Marker for transversal (hiro) */}
                  <a-marker preset="hiro" id="marker-transversal">
                    <a-entity id="wave-transversal" position="0 0 0" rotation="0 0 0" sine-wave="amplitude: 0.5; frequency:1; type: transversal"></a-entity>
                  </a-marker>

                  {/* Marker for longitudinal (kanji) */}
                  <a-marker preset="kanji" id="marker-longitudinal">
                    <a-entity id="wave-longitudinal" position="0 0 0" rotation="0 0 0" sine-wave="amplitude: 0.5; frequency:1; type: longitudinal"></a-entity>
                  </a-marker>

                  <a-entity camera></a-entity>
                </a-scene>
              </div>
            </div>
          </div>

          {/* Right: Controls */}
          <div className="bg-white shadow rounded p-4 flex flex-col gap-4">
            <div>
              <label className="block text-sm font-medium text-gray-700">Pilih Gelombang</label>
              <div className="mt-2 flex items-center gap-3">
                <button
                  className={`px-3 py-1 rounded ${waveType === "transversal" ? "bg-blue-600 text-white" : "bg-gray-200"}`}
                  onClick={() => setWaveType("transversal")}
                >
                  Transversal
                </button>
                <button
                  className={`px-3 py-1 rounded ${waveType === "longitudinal" ? "bg-blue-600 text-white" : "bg-gray-200"}`}
                  onClick={() => setWaveType("longitudinal")}
                >
                  Longitudinal
                </button>
              </div>
            </div>

            <div>
              <label className="block text-sm font-medium text-gray-700">Amplitudo: {amplitude.toFixed(2)}</label>
              <input
                type="range"
                min="0"
                max="1.5"
                step="0.01"
                value={amplitude}
                onChange={(e) => setAmplitude(parseFloat(e.target.value))}
                className="w-full"
              />
            </div>

            <div>
              <label className="block text-sm font-medium text-gray-700">Frekuensi: {frequency.toFixed(2)}</label>
              <input
                type="range"
                min="0.1"
                max="5"
                step="0.1"
                value={frequency}
                onChange={(e) => setFrequency(parseFloat(e.target.value))}
                className="w-full"
              />
            </div>

            <div>
              <label className="block text-sm font-medium text-gray-700">Ringkasan Materi</label>
              <div className="mt-2 text-sm text-gray-700">
                <p>
                  Gelombang mekanik adalah gangguan yang merambat melalui medium. Gelombang transversal memiliki arah getar yang tegak lurus terhadap arah rambat, sedangkan pada gelombang longitudinal arah getar sejajar dengan arah rambat.
                </p>
              </div>
            </div>

            <div>
              <label className="block text-sm font-medium text-gray-700">Contoh Aplikasi</label>
              <ul className="list-disc pl-5 text-sm text-gray-700 mt-2">
                <li>Gelombang bunyi dalam komunikasi (sonar, ultrasonics).</li>
                <li>Gelombang pada tali & instrumen musik.</li>
                <li>Gelombang seismik yang dipelajari dalam seismologi.</li>
              </ul>
            </div>

            <div>
              <label className="block text-sm font-medium text-gray-700">Latihan Singkat</label>
              <div className="mt-2 text-sm text-gray-700">
                <p>1) Jika frekuensi dinaikkan dan medium tetap, apa yang terjadi pada panjang gelombang?</p>
                <div className="mt-2 grid grid-cols-1 gap-2">
                  <button className="px-3 py-1 rounded bg-gray-200 text-left">A. Panjang gelombang meningkat</button>
                  <button className="px-3 py-1 rounded bg-gray-200 text-left">B. Panjang gelombang tetap</button>
                  <button className="px-3 py-1 rounded bg-gray-200 text-left">C. Panjang gelombang menurun</button>
                </div>
              </div>
            </div>

          </div>
        </div>

        <p className="text-xs text-gray-500 mt-3">Catatan: Aplikasi ini adalah prototipe WebAR marker-based. Untuk penggunaan di kelas, cetak marker (mis. kode 'hiro'/'kanji') dan arahkan kamera perangkat pada marker tersebut.</p>
      </div>
    </div>
  );
}
